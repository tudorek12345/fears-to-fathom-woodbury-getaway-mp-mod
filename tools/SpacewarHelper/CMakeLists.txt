cmake_minimum_required(VERSION 3.20)
project(WoodburySpacewar LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(SPACEWAR_BUILD_HELPER "Build the Steamworks helper (official SDK required)" ON)
set(STEAMWORKS_SDK "" CACHE PATH "Official Steamworks sdk folder containing public and redistributable_bin")
enable_testing()
add_executable(SpacewarCoreTests tests.cpp)
add_test(NAME SpacewarCoreTests COMMAND SpacewarCoreTests)
if(MSVC)
  set_property(TARGET SpacewarCoreTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(SpacewarCoreTests PRIVATE /W4 /WX)
endif()
if(SPACEWAR_BUILD_HELPER)
  if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Spacewar helper requires Windows x64")
  endif()
  foreach(required public/steam/steam_api.h public/steam/isteamnetworkingmessages.h redistributable_bin/win64/steam_api64.lib redistributable_bin/win64/steam_api64.dll)
    if(NOT EXISTS "${STEAMWORKS_SDK}/${required}")
      message(FATAL_ERROR "Missing ${required}. Set STEAMWORKS_SDK to the official SDK's sdk directory.")
    endif()
  endforeach()
  add_executable(Woodbury.Spacewar main.cpp)
  add_custom_command(TARGET Woodbury.Spacewar POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${STEAMWORKS_SDK}/redistributable_bin/win64/steam_api64.dll" "$<TARGET_FILE_DIR:Woodbury.Spacewar>")
  add_test(NAME SpacewarOutputIsolation COMMAND powershell.exe -NoProfile -ExecutionPolicy Bypass
    -File "${CMAKE_CURRENT_SOURCE_DIR}/TestOutputIsolation.ps1" -HelperPath "$<TARGET_FILE:Woodbury.Spacewar>")
  target_include_directories(Woodbury.Spacewar SYSTEM PRIVATE "${STEAMWORKS_SDK}/public")
  target_link_libraries(Woodbury.Spacewar PRIVATE "${STEAMWORKS_SDK}/redistributable_bin/win64/steam_api64.lib")
  target_compile_definitions(Woodbury.Spacewar PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  if(MSVC)
    target_compile_options(Woodbury.Spacewar PRIVATE /W4)
    set_property(TARGET Woodbury.Spacewar PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
  install(TARGETS Woodbury.Spacewar RUNTIME DESTINATION Spacewar)
  install(FILES "${STEAMWORKS_SDK}/redistributable_bin/win64/steam_api64.dll" DESTINATION Spacewar)
  # Development App ID hint only in the helper directory, never beside the game executable.
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/steam_appid.txt" "480\n")
  install(FILES "${CMAKE_CURRENT_BINARY_DIR}/steam_appid.txt" DESTINATION Spacewar)
endif()
